#include <cmath>

// Rubi: mic clean-up filters (4th-order Butterworth high-pass 130 Hz removes mains/USB hum,
// 4th-order low-pass 7 kHz removes the very high whine above the speech band).
namespace {
struct RubiBiquad {
    float b0 = 1, b1 = 0, b2 = 0, a1 = 0, a2 = 0, z1 = 0, z2 = 0;
    void Design(bool high, float fc, float fs, float q) {
        float w = 2.0f * 3.14159265f * fc / fs;
        float c = cosf(w), s = sinf(w), al = s / (2.0f * q);
        float a0 = 1.0f + al;
        if (high) { b0 = (1 + c) / 2; b1 = -(1 + c); b2 = (1 + c) / 2; }
        else      { b0 = (1 - c) / 2; b1 = 1 - c;    b2 = (1 - c) / 2; }
        a1 = -2.0f * c; a2 = 1.0f - al;
        b0 /= a0; b1 /= a0; b2 /= a0; a1 /= a0; a2 /= a0;
        z1 = z2 = 0;
    }
    inline float Run(float x) {
        float y = b0 * x + z1;
        z1 = b1 * x - a1 * y + z2;
        z2 = b2 * x - a2 * y;
        return y;
    }
};
RubiBiquad g_hp[2], g_lp[2];
bool g_filters_ready = false;
}  // namespace
