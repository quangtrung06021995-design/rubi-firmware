    void InitializeTools() {
        McpServer::GetInstance().AddTool("self.weather.get_current",
            "Get the real current weather and today's forecast for the place where the device is now. "
            "Always use this tool when the user asks about weather, temperature, rain or whether to bring an umbrella or coat. "
            "Read the returned text to the user as is; do not guess the location.",
            PropertyList(), [](const PropertyList&) -> ReturnValue {
                auto http = Board::GetInstance().GetNetwork()->CreateHttp(0);
                if (!http->Open("GET", "https://rubi-firmware.vercel.app/api/weather")) {
                    return std::string("Khong lay duoc thoi tiet luc nay.");
                }
                std::string body = http->ReadAll();
                http->Close();
                return body;
            });
    }
