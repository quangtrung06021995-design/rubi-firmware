    void InitializeTools() {
        McpServer::GetInstance().AddTool("self.weather.get_current",
            "Get the real current weather and today's forecast for the place where the device is now. "
            "Always use this tool when the user asks about weather, temperature, rain or whether to bring an umbrella or coat. "
            "Read the returned text to the user as is; do not guess the location.",
            PropertyList(), [](const PropertyList&) -> ReturnValue {
                auto http = Board::GetInstance().GetNetwork()->CreateHttp(0);
                if (!http->Open("GET", "https://rubi-firmware.vercel.app/api/weather")) {
                    return std::string("Khong lay duoc thoi tiet luc nay.");
                }
                std::string body = http->ReadAll();
                http->Close();
                return body;
            });

        McpServer::GetInstance().AddTool("self.clock.get_current_time",
            "Get the real current local date and time of the device (the correct time zone for where the device is). "
            "Always use this tool when the user asks what time it is, what day or date it is, or anything about the current time. "
            "Never guess the time. Read the returned text to the user as is.",
            PropertyList(), [](const PropertyList&) -> ReturnValue {
                time_t now = time(nullptr);
                struct tm t;
                localtime_r(&now, &t);
                static const char* kDays[] = {"Chủ nhật", "thứ Hai", "thứ Ba", "thứ Tư", "thứ Năm", "thứ Sáu", "thứ Bảy"};
                const char* period = "sáng";
                if (t.tm_hour < 5) period = "đêm";
                else if (t.tm_hour < 11) period = "sáng";
                else if (t.tm_hour < 13) period = "trưa";
                else if (t.tm_hour < 18) period = "chiều";
                else if (t.tm_hour < 22) period = "tối";
                else period = "đêm";
                int h12 = t.tm_hour % 12;
                if (h12 == 0) h12 = 12;
                char buf[200];
                snprintf(buf, sizeof(buf),
                         "Bây giờ là %d giờ %02d phút %s (%02d:%02d theo giờ 24), %s, ngày %d tháng %d năm %d.",
                         h12, t.tm_min, period, t.tm_hour, t.tm_min, kDays[t.tm_wday],
                         t.tm_mday, t.tm_mon + 1, t.tm_year + 1900);
                return std::string(buf);
            });
    }
